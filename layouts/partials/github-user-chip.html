{{- $username := .username -}} {{- $context := .context -}} {{- if not $username -}} {{- errorf "github-user-chip partial requires a 'username' parameter" -}} {{- end -}} {{- $apiURL := printf "https://api.github.com/users/%s" $username -}} {{- $response := resources.GetRemote $apiURL -}} {{- $userData := $response | transform.Unmarshal -}} {{- if $userData -}} {{- $avatarURL := $userData.avatar_url -}} {{- $displayName := $userData.name | default $userData.login -}} {{- $profileURL := $userData.html_url -}} {{ $displayName }} {{- else -}} {{- warnf "Failed to fetch GitHub user data for username: %s" $username -}} {{ partial "icon.html" (dict "name" "github" "class" "w-6 h-6 text-black dark:text-white flex-shrink-0") }} {{ $username }} {{- end -}}
